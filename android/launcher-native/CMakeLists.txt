cmake_minimum_required(VERSION 3.24)
project(wwe13_android_launcher LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(WWE13_REXGLUE_THIRDPARTY ${CMAKE_CURRENT_LIST_DIR}/../../sdk/thirdparty)
add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../../game/launcher2/core core)
target_include_directories(wwe13_launcher_core PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/gen)
add_library(wwe13launcher SHARED jni_launcher.cpp)
target_link_libraries(wwe13launcher PRIVATE wwe13_launcher_core log android)
target_link_options(wwe13launcher PRIVATE -Wl,--no-undefined -Wl,-Bsymbolic)
